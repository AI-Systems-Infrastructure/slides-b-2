\begin{frame}{Container in computers}
  A piece of software relies on its code, configuration, runtime environments, file system layout, runtime states, \dots to \textit{work as expected}.

  \begin{block}{Software deployment without containers}
    \begin{itemize}
      \item Setting up everything right on each computer the software is deployed to is tedious and prone to error
      \item Multiple pieces of software running on the same computer might interfere with each other
    \end{itemize}
  \end{block}

  \begin{block}{Software deployment with containers}
    \begin{itemize}
      \item A container packs everything a piece of software needs in one self-contained package
      \item A container provides an isolated environment for software to run without interference
    \end{itemize}
  \end{block}

\note{
}
\end{frame}


\begin{frame}{Container image vs. container instance}
  \begin{itemize}
    \item \textbf{Container image:} A static, read-only blueprint of everything needed to run a program
    \item \textbf{Container instance:} A running program initialized from an image
  \end{itemize}

  % TODO: Embed two images of the same PS5 game, left is the game disc, right is a running game instance.

\note{
}
\end{frame}


\begin{frame}{Container vs. virtual machine}
  A \textbf{virtual machine (VM)} is an alternative approach to the same problem.

  % TODO: Embed a diagram comparing VM and container.

  \begin{itemize}
    \item \textbf{VM:} Each instance has its own isolated kernel and operating system, better isolation
    \item \textbf{Container:} All instances share the kernel of the host computer, more lightweight
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}{Layered structure of containers}{Image layer and container layer}
  \begin{itemize}
    \item \textbf{In a container image:} Stacks of image layers, each containing one part of what a program needs
    \item \textbf{In a container instance:} A writable container layer on top of the read-only image layers for instance-specific states
  \end{itemize}

  % TODO: Embed a diagram showing the layer structure
\note{
}
\end{frame}


\begin{frame}[fragile]{Layered structure of containers}{Image layer}
  Image layers in a typical container image for an AI system:

    \begin{minted}{text}
4. Application code
3. Python packages
2. Python runtime
1. Linux base
    \end{minted}

    \begin{itemize}
      \item The same layer is shared across different container images on a host computer
      \item Modification to one layer also changes other layers on top of it
    \end{itemize}

}
\end{frame}


\begin{frame}{Layered structure of containers}{Container layer}
  \begin{itemize}
    \item Created at the initialization of a container instance
    \item Holds anything the running program writes
    \item Discarded at the termination of the container instance
  \end{itemize}

\section{Using Containers}


\begin{frame}[fragile]{Find a container image}
  \begin{minted}{bash}
docker search python
  \end{minted}
  \begin{cmdoutput}
NAME              DESCRIPTION                                       STARS     OFFICIAL
python            Python is an interpreted, interactive, objec...   10539     [OK]
circleci/python   Python is an interpreted, interactive, objec...   117
cimg/python                                                         25
  \end{cmdoutput}

  \begin{itemize}
    \item Searches Docker Hub for container images matching a keyword
    \item Official container images are marked \texttt{[OK]} in the \texttt{OFFICIAL} column
    \item Browse \texttt{hub.docker.com} for the available tags and documentation of a container image
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Pull a container image}
  \begin{minted}{bash}
docker pull python:3.13
  \end{minted}
  \begin{cmdoutput}
3.13: Pulling from library/python
6c3776672264: Pull complete
9f903ac3ecde: Pull complete
...
Digest: sha256:94e0b775113cf19d52f692d34d6084590c6d68f3f95a73a5071f503890d167c9
Status: Downloaded newer image for python:3.13
docker.io/library/python:3.13
  \end{cmdoutput}

  \begin{itemize}
    \item \textbf{Format:} \texttt{<repository>:<tag>}
    \item \textbf{Tag:} Picks a version of the container image, defaults to \texttt{latest} if omitted
    \item Pinning a specific tag is safer than relying on \texttt{latest}
    \item Smaller variants of the same version exist, like \texttt{python:3.13-slim} and \texttt{python:3.13-alpine}
  \end{itemize}

\note{
}
\end{frame}

\begin{frame}[fragile]{Run a container instance}{Detached mode}
  \begin{minted}{bash}
docker run -d python:3.13 sleep 3600
  \end{minted}

  \begin{itemize}
    \item \texttt{-d} runs the container instance in the background and returns our terminal right away
    \item Suited for long-running services
  \end{itemize}

\end{frame}


\begin{frame}[fragile]{Name a container instance}
  \begin{minted}{bash}
docker run --name my-app -d python:3.13 sleep 3600
  \end{minted}

  \begin{itemize}
    \item By default, each container instance gets an autogenerated ID and a random two-word name like \texttt{eager\_tesla}
    \item \texttt{-{}-name} gives the container instance a name of our choice
  \end{itemize}

\begin{frame}{Talking to the host computer}
  A container instance is \textbf{isolated from the host computer} by design:
  \begin{itemize}
    \item Files written inside stay inside and are discarded after it is removed
    \item Ports are not reachable from outside
    \item The host's environment variables are invisible to it
  \end{itemize}

  \vspace{1.0em}
  We explicitly open holes when we want the container instance to interact with the host:
  \begin{itemize}
    \item \textbf{Ports:} Reach services running inside the container instance
    \item \textbf{Volumes:} Share files between the host and the container instance
    \item \textbf{Environment variables:} Pass configuration from the host into the container instance
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Port mapping}
  \begin{minted}{bash}
docker run -p 8000:8000 python:3.13 python -m http.server 8000
  \end{minted}
  \begin{cmdoutput}
172.17.0.1 - - [19/Oct/2026 10:15:32] "GET / HTTP/1.1" 200 -
  \end{cmdoutput}

  \begin{itemize}
    \item \textbf{Format:} \texttt{-p <host\_port>:<container\_port>}
    \item Visit \texttt{http://127.0.0.1:8000} on the host to reach the server inside the container instance
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Port mapping}{Different host and container ports}
  \begin{minted}{bash}
docker run -p 3000:8000 python:3.13 python -m http.server 8000
  \end{minted}
  \begin{cmdoutput}
172.17.0.1 - - [19/Oct/2026 10:18:04] "GET / HTTP/1.1" 200 -
  \end{cmdoutput}

  \begin{itemize}
    \item The two ports do not have to match
    \item The server still listens on port 8000 inside the container instance, but appears as port 3000 on the host
    \item Useful when several container instances use the same internal port
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Volume mapping}
  \begin{minted}{bash}
docker run -v $(pwd):/app python:3.13 ls /app
  \end{minted}
  \begin{cmdoutput}
README.md
client.py
  \end{cmdoutput}

  \begin{itemize}
    \item \textbf{Format:} \texttt{-v <host\_path>:<container\_path>}
    \item The current directory appears as \texttt{/app} inside the container instance
    \item Changes on either side are visible to the other
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}{Volume mapping}{Common uses}
  \begin{itemize}
    \item \textbf{Feed files in:} Pass code or configuration into a container instance without baking it into the container image
    \item \textbf{Persist data:} Keep data, like a database file, beyond the lifetime of a container instance
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Environment variables}
  \begin{minted}{bash}
docker run -e OPENAI_API_KEY=sk-abc... python:3.13 \
    python -c "import os; print(os.getenv('OPENAI_API_KEY'))"
  \end{minted}
  \begin{cmdoutput}
sk-abc...
  \end{cmdoutput}

  \begin{itemize}
    \item \textbf{Format:} \texttt{-e <KEY>=<value>}
    \item A container instance does not see the host's environment variables, so we pass them in explicitly
  \end{itemize}

\note{
}
\end{frame}


\begin{frame}[fragile]{Environment variables}{From a file}
  \begin{minted}{bash}
docker run --env-file .env python:3.13 \
    python -c "import os; print(dict(os.environ))"
  \end{minted}
  \begin{cmdoutput}
{'PATH': '/usr/local/bin:...', ..., 'OPENAI_API_KEY': 'sk-abc...', ...}
  \end{cmdoutput}

  \begin{itemize}
    \item Reads all variables from a file, convenient when there are many
    \item Same \texttt{.env} format as Module A.2: one \texttt{KEY=value} per line
  \end{itemize}

\note{
}
\end{frame}
